# Ojin SDK (high-level Speech-To-Video client). Provides ojin.stv.OjinSTVClient,

ojin-client[stv]>=0.6.7

# Pipecat + the services this agent uses. Verified against pipecat-ai 1.3.0.
#   runner    -> the dev runner + prebuilt browser UI (http://localhost:7860/client)
#   webrtc    -> local WebRTC call in the browser (the default, no account needed)
#   daily     -> optional: join a Daily room instead (`-t daily`, needs an account)
#   silero    -> on-device voice-activity detection (turn-taking + barge-in)
#   deepgram  -> speech-to-text
#   openai    -> the LLM
#   cartesia  -> text-to-speech
pipecat-ai[runner,webrtc,daily,silero,deepgram,google,cartesia]~=1.3.0

# That's it — no extra logging or dotenv dependency. ojin.load_env() reads the
